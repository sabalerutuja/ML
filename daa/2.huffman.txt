#2.Title: Write a program to implement Huffman Encoding using a greedy strategy.


import heapq 
# Node class for Huffman Tree 
class Node: 
    def __init__(self, freq, char=None, left=None, right=None): 
        self.freq = freq 
        self.char = char 
        self.left = left 
        self.right = right 
 
    # Compare nodes based on frequency 
    def __lt__(self, other): 
        return self.freq < other.freq 
 
# Function to generate Huffman Codes 
def generate_codes(node, code="", huffman_code={}): 
    if node is None: 
        return 
 
    # If leaf node 
    if node.char is not None: 
        huffman_code[node.char] = code 
 
    generate_codes(node.left, code + "0", huffman_code) 
    generate_codes(node.right, code + "1", huffman_code) 
 
    return huffman_code 
 
# Character frequencies 
chars = { 
    'A': 6, 
    'B': 7, 
    'C': 2, 
    'D': 23, 
    'E': 22, 
    'F': 47 
} 
 
# Create priority queue 
heap = [] 
 
for char, freq in chars.items(): 
    heapq.heappush(heap, Node(freq, char)) 
 
# Build Huffman Tree 
while len(heap) > 1: 
    left = heapq.heappop(heap) 
    right = heapq.heappop(heap) 
 
    merged = Node(left.freq + right.freq, left=left, right=right) 
 
    heapq.heappush(heap, merged) 
 
# Root node 
root = heap[0] 
 
# Generate Huffman Codes 
codes = generate_codes(root) 
print("Huffman Codes") 
print("---------------------------") 
for char in sorted(codes): 
    print(char, ":", codes[char]) 
# Calculate Average Code Length 
total_frequency = sum(chars.values()) 
average = 0 
print("\nEncoded Length of Each Character") 
print("----------------------------------") 
for char in sorted(chars): 
    bits = len(codes[char]) * chars[char] 
average += bits 
print(char, "Frequency =", chars[char], 
"Code =", codes[char], 
"Bits =", bits) 
print("\nAverage Code Length =", round(average / total_frequency, 2), "bits") 
print("Total Huffman Encoded Message Length =", average, "bits")
